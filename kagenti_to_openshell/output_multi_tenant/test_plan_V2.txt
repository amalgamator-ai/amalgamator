2.3 Test Helper: make_user()
----------------------------

  Creates a tonic::Request with Principal::User in extensions.
  Placed alongside test_server_state_with_ownership.

  fn make_user(subject: &str, roles: Vec<&str>) -> Principal {
      Principal::User(UserPrincipal {
          identity: Identity {
              subject: subject.to_string(),
              display_name: Some(subject.to_string()),
              roles: roles.into_iter().map(String::from).collect(),
              scopes: vec![],
              groups: vec![],     // NEW field from Step 1
              provider: IdentityProvider::Oidc,
          },
      })
  }

  fn make_admin() -> Principal {
      make_user("admin-uuid", vec!["openshell-admin", "openshell-user"])
  }

  fn make_regular_user(subject: &str) -> Principal {
      make_user(subject, vec!["openshell-user"])
  }

  fn request_with_principal<T>(payload: T, principal: &Principal) -> Request<T> {
      let mut request = Request::new(payload);
      request.extensions_mut().insert(principal.clone());
      request
  }

4.1 extract_groups() Method Tests
---------------------------------

  TEST: extract_groups_from_flat_array_claim
    Setup:  Deserialize OidcClaims from JSON:
            {"sub": "user-1", "groups": ["/team1", "/admins"]}
    Action: claims.extract_groups("groups")
    Expect: claims.groups == ["/team1", "/admins"]
    Why:    Standard Keycloak groups claim is a flat JSON array.

  TEST: extract_groups_empty_when_claim_absent
    Setup:  Deserialize from: {"sub": "user-1"}
    Action: claims.extract_groups("groups")
    Expect: claims.groups == []
    Why:    Missing claim yields empty vec (no panic). Matches
            extract_roles behavior when roles_claim path doesn't exist.

  TEST: extract_groups_from_nested_claim_path
    Setup:  {"sub": "user-1", "custom": {"team_groups": ["eng", "ops"]}}
    Action: claims.extract_groups("custom.team_groups")
    Expect: claims.groups == ["eng", "ops"]
    Why:    Nested dot-path support, same as extract_roles.

  TEST: extract_groups_empty_when_claim_is_not_array
    Setup:  {"sub": "user-1", "groups": "single-group-string"}
    Action: claims.extract_groups("groups")
    Expect: claims.groups == []
    Why:    Non-array value yields empty (extract_roles returns empty
            when the claim is a string, not an array).

  TEST: extract_groups_skips_non_string_elements
    Setup:  {"sub": "user-1", "groups": ["/team1", 42, true, "/team2"]}
    Action: claims.extract_groups("groups")
    Expect: claims.groups == ["/team1", "/team2"]
    Why:    filter_map(|v| v.as_str()) pattern filters non-strings.

  TEST: extract_groups_deeply_nested_path
    Setup:  {"sub": "u1", "a": {"b": {"c": ["g1", "g2"]}}}
    Action: claims.extract_groups("a.b.c")
    Expect: claims.groups == ["g1", "g2"]
    Why:    Multi-level dot-path traversal.

  TEST: extract_groups_partial_path_returns_empty
    Setup:  {"sub": "u1", "a": {"b": "not-an-object"}}
    Action: claims.extract_groups("a.b.c")
    Expect: claims.groups == []
    Why:    Path traversal hits non-object before reaching the leaf.

5.1 sanitize_label_value() Tests

  Purpose: Kubernetes label values must match [a-z0-9A-Z._-]{0,63},

  strings (e.g., email addresses, UUIDs) into valid label values.

  TEST: sanitize_email_replaces_at_and_dots
    Input:  "alice@example.com"
    Expect: "alice_example.com" or equivalent valid label
    Why:    @ is not valid in K8s labels; must be replaced.

  TEST: sanitize_empty_string
    Input:  ""
    Expect: "" (or a defined sentinel)
    Why:    Edge case: empty subject should not panic.

  TEST: sanitize_long_string_truncated_to_63
    Input:  "a".repeat(100)
    Expect: Result.len() <= 63
    Why:    K8s label value max is 63 characters.

  TEST: sanitize_special_characters_replaced
    Input:  "user/name:admin"
    Expect: All / and : replaced with valid chars (underscore or dash)
    Why:    / and : are common in identity subjects but invalid in labels.

  TEST: sanitize_leading_trailing_non_alnum_stripped
    Input:  ".leading-dot-"
    Expect: "leading-dot" or similar (no leading/trailing . or -)
    Why:    K8s labels must start/end with alphanumeric.

  TEST: sanitize_unicode_handled

  TEST: sanitize_already_valid_label_unchanged

  TEST: sanitize_numeric_only

  TEST: sanitize_single_char
