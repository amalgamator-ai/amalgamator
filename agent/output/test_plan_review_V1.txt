  describe("IterationBudget", () => {

    --- 1.1 Constructor & Defaults ---

    describe("constructor", () => {

      it("creates a budget with specified maxTotal and zero used", () => {
        Verify: new IterationBudget(90) has:
          - maxTotal === 90
          - used === 0
          - remaining === 90
      });

      it("accepts maxTotal of 1 (minimum useful budget)", () => {
        Verify: new IterationBudget(1) has:
          - maxTotal === 1
          - remaining === 1
          - consume() returns true
          - consume() returns false (budget exhausted after 1)
      });

      it("handles maxTotal of 0 (immediately exhausted)", () => {
        Verify: new IterationBudget(0) has:
          - maxTotal === 0
          - remaining === 0
          - consume() returns false immediately
      });
    });

    --- 1.2 consume() Method ---

    describe("consume", () => {

      it("returns true and increments used when under budget", () => {
        Setup: budget = new IterationBudget(5)
        Action: Call consume()
        Verify:
          - Returns true
          - budget.used === 1
          - budget.remaining === 4
      });

      it("returns true for each consume up to maxTotal", () => {
        Setup: budget = new IterationBudget(3)
        Action: Call consume() 3 times
        Verify:
          - All 3 return true
          - budget.used === 3
          - budget.remaining === 0
      });

      it("returns false when budget is exhausted", () => {
        Setup: budget = new IterationBudget(3)
        Action: Consume 3 times (exhaust), then consume once more
        Verify:
          - 4th consume returns false
          - budget.used === 3 (unchanged)
          - budget.remaining === 0 (unchanged)
      });

      it("returns false on repeated calls after exhaustion", () => {
        Setup: budget = new IterationBudget(2)
        Action: Consume 2 (exhaust), then consume 3 more times
        Verify: All 3 extra calls return false; used stays at 2
      });
    });

    --- 1.3 refund() Method ---

    describe("refund", () => {

      it("decrements used count by 1", () => {
        Setup: budget = new IterationBudget(5), consume twice
        Action: refund()
        Verify:
          - budget.used === 1
          - budget.remaining === 4
      });

      it("restores consume ability after exhaustion", () => {
        Setup: budget = new IterationBudget(2), consume 2 (exhausted)
        Action: refund()
        Verify:
          - budget.remaining === 1
          - consume() returns true
      });

      it("floors used at 0 (refund on fresh budget is no-op)", () => {
        Setup: budget = new IterationBudget(5)
        Action: refund() (without any prior consume)
        Verify:
          - budget.used === 0 (not -1)
          - budget.remaining === 5 (unchanged)
      });

      it("floors used at 0 after multiple refunds beyond usage", () => {
        Setup: budget = new IterationBudget(3), consume once
        Action: refund() 5 times
        Verify:
          - budget.used === 0 (never goes negative)
          - budget.remaining === 3
      });
    });

    --- 1.4 used / remaining Properties ---

    describe("used and remaining", () => {

      it("used reflects consume and refund accurately", () => {
        Setup: budget = new IterationBudget(5)
        Action sequence: consume, consume, refund, consume, refund, refund
        Verify: Each step matches expected used count: 1, 2, 1, 2, 1, 0
      });

      it("remaining equals maxTotal minus used", () => {
        Setup: budget = new IterationBudget(5)
        Action sequence: consume 3, refund 1, consume 3
        Verify: remaining at each step: 2, 3, 0
      });

      it("remaining never goes below 0", () => {
        Setup: budget = new IterationBudget(2)
        Action: consume 2, then try consume 5 more
        Verify: remaining stays at 0 after exhaustion
      });
    });

    --- 1.5 Interleaved Consume/Refund Sequences ---

    describe("interleaved sequences", () => {

      it("handles alternating consume-refund pattern without drift", () => {
        Setup: budget = new IterationBudget(3)
        Action: 10 cycles of [consume, refund]
        Verify:
          - Each consume returns true
          - After all cycles: used=0, remaining=3
      });

      it("tracks state correctly through complex consume/refund sequence", () => {
        Setup: budget = new IterationBudget(5)
        Action:
          - consume x3 -> used=3
          - refund x1 -> used=2
          - consume x2 -> used=4
          - refund x4 -> used=0
          - consume x5 -> used=5 (exhausted)
          - consume x1 -> false, used=5
          - refund x1 -> used=4
          - consume x1 -> true, used=5
        Verify: Each intermediate state
      });
    });

    [FIX #1] REMOVED: Section 1.6 "snapshot() method"

    The Hermes IterationBudget (62 lines) does NOT have a snapshot() method.
    It has: __init__, consume, refund, used property, remaining property.
    The port plan Step 13 reads used/remaining directly from the budget to
    populate result metadata. There is no need for a separate snapshot() API.

    If snapshot() is added as an OpenClaw convenience, it should be a separate
    implementation decision, not part of the core port. Tests for Step 13
    metadata are covered in FILE 4 (integration tests) by asserting
    result.meta.iterationBudget values match the budget's used/remaining.
  });

  [FIX #2] CORRECTED: File location changed from separate file to EXISTING file.

  V1 ERROR: V1 proposed creating a new file:
    src/config/zod-schema.iteration-budget.test.ts

  CORRECTION: OpenClaw does NOT create separate test files per schema section.
  All agent defaults schema tests go in the EXISTING file:
    src/config/zod-schema.agent-defaults.test.ts

  Evidence: All 30+ zod-schema test files match their source file name.
  The pattern is zod-schema.agent-defaults.ts -> zod-schema.agent-defaults.test.ts.
  Adding a new section-specific test file would violate this convention.

  [FIX #3] CORRECTED: Default value test assumptions.

  V1 ERROR: V1 assumed that Zod schema fields use .default() to apply
  default values during parsing. Section 2.4 tested:
    AgentDefaultsSchema.parse({ iterationBudget: { enabled: true } })
    -> result.iterationBudget.maxIterations === 90

  CORRECTION: AgentDefaultsSchema uses .optional() NOT .default() for all
  fields. AgentDefaultsSchema.parse({}) returns `undefined` (the whole
  schema is .optional()). Individual fields return `undefined` when not
  specified, not default values. Default merging happens in config resolution
  code (resolveAgentConfig), not in the Zod schema.

  TWO VALID APPROACHES:

    APPROACH A: If the implementation adds .default() to the iterationBudget
    sub-schema fields (z.number().int().min(1).default(90)), then the V1
    default tests would work. But this would be UNUSUAL for OpenClaw.

    APPROACH B (MATCHES CONVENTION): Keep fields .optional() in the schema,
    apply defaults in resolveAgentConfig or resolveIterationBudgetConfig.
    In this case, default value tests go in FILE 3 (config resolution), not
    here.

  THE CORRECTED PLAN BELOW ASSUMES APPROACH B (matching OpenClaw convention).
  If the implementation uses Approach A, re-enable the default tests here.
