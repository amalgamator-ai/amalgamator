Purpose:      Unit tests for the IterationBudget class
Source file:  src/agents/iteration-budget.ts (new file, Step 1)
Hermes analog: tests/run_agent/test_iteration_budget_race.py

--- Why no thread-safety tests ---

  Hermes's test_iteration_budget_used_is_thread_safe uses 10 threads x 200
  ops with ThreadPoolExecutor to verify the threading.Lock protects
  concurrent access. Node.js is single-threaded (no data races possible),
  so the TypeScript IterationBudget does NOT need a Lock/Mutex and the
  thread-safety test is correctly EXCLUDED.

--- Imports ---

  import { describe, it, expect } from "vitest";
  import { IterationBudget } from "./iteration-budget.js";

--- Test Structure ---

  describe("IterationBudget", () => {

    // === 1.1 Constructor & Defaults ===

    describe("constructor", () => {

      it("creates a budget with specified maxTotal and zero used", () => {
        Arrange: const budget = new IterationBudget(90)
        Assert:
          expect(budget.maxTotal).toBe(90)
          expect(budget.used).toBe(0)
          expect(budget.remaining).toBe(90)
      });

      it("accepts maxTotal of 1 (minimum useful budget)", () => {
        Arrange: const budget = new IterationBudget(1)
        Assert:
          expect(budget.maxTotal).toBe(1)
          expect(budget.remaining).toBe(1)
          expect(budget.consume()).toBe(true)
          expect(budget.consume()).toBe(false)
      });

      it("handles maxTotal of 0 (immediately exhausted)", () => {
        Arrange: const budget = new IterationBudget(0)
        Assert:
          expect(budget.maxTotal).toBe(0)
          expect(budget.remaining).toBe(0)
          expect(budget.consume()).toBe(false)
      });
    });

    // === 1.2 consume() ===

    describe("consume", () => {

      it("returns true and increments used when under budget", () => {
        Arrange: const budget = new IterationBudget(5)
        Act:     const result = budget.consume()
        Assert:
          expect(result).toBe(true)
          expect(budget.used).toBe(1)
          expect(budget.remaining).toBe(4)
      });

      it("returns true for each consume up to maxTotal", () => {
        Arrange: const budget = new IterationBudget(3)
        Act:     const results = [budget.consume(), budget.consume(), budget.consume()]
        Assert:
          expect(results).toEqual([true, true, true])
          expect(budget.used).toBe(3)
          expect(budget.remaining).toBe(0)
      });

      it("returns false when budget is exhausted", () => {
        Arrange: const budget = new IterationBudget(3)
        Act:     consume 3 times, then const fourth = budget.consume()
        Assert:
          expect(fourth).toBe(false)
          expect(budget.used).toBe(3)   // unchanged
          expect(budget.remaining).toBe(0)
      });

      it("returns false on repeated calls after exhaustion", () => {
        Arrange: const budget = new IterationBudget(2)
        Act:     consume 2 (exhaust), then consume 3 more
        Assert:
          All 3 extra calls return false
          expect(budget.used).toBe(2)   // stays at 2
      });
    });

    // === 1.3 refund() ===

    describe("refund", () => {

      it("decrements used count by 1", () => {
        Arrange: const budget = new IterationBudget(5)
        Act:     budget.consume(); budget.consume(); budget.refund()
        Assert:
          expect(budget.used).toBe(1)
          expect(budget.remaining).toBe(4)
      });

      it("restores consume ability after exhaustion", () => {
        Arrange: const budget = new IterationBudget(2)
        Act:     consume 2 (exhausted), then refund()
        Assert:
          expect(budget.remaining).toBe(1)
          expect(budget.consume()).toBe(true)
      });

      it("floors used at 0 (refund on fresh budget is no-op)", () => {
        Arrange: const budget = new IterationBudget(5)
        Act:     budget.refund()
        Assert:
          expect(budget.used).toBe(0)
          expect(budget.remaining).toBe(5)
      });

      it("floors used at 0 after multiple refunds beyond usage", () => {
        Arrange: const budget = new IterationBudget(3)
        Act:     budget.consume(); for (let i = 0; i < 5; i++) budget.refund()
        Assert:
          expect(budget.used).toBe(0)
          expect(budget.remaining).toBe(3)
      });
    });

    // === 1.4 used / remaining Properties ===

    describe("used and remaining", () => {

      it("used reflects consume and refund accurately", () => {
        Arrange: const budget = new IterationBudget(10)
        Act + Assert:
          expect(budget.used).toBe(0)
          budget.consume(); expect(budget.used).toBe(1)
          budget.consume(); expect(budget.used).toBe(2)
          budget.refund();  expect(budget.used).toBe(1)
          budget.refund();  expect(budget.used).toBe(0)
      });

      it("remaining equals maxTotal minus used", () => {
        Arrange: const budget = new IterationBudget(5)
        Act + Assert:
          expect(budget.remaining).toBe(5)
          budget.consume(); expect(budget.remaining).toBe(4)
          budget.consume(); budget.consume(); expect(budget.remaining).toBe(2)
          budget.refund(); expect(budget.remaining).toBe(3)
      });

      it("remaining never goes below 0", () => {
        Arrange: const budget = new IterationBudget(2)
        Act:     consume 2, then try consume 5 more
        Assert:  expect(budget.remaining).toBe(0) after each extra consume
      });
    });

    // === 1.5 Interleaved Consume/Refund Sequences ===

    describe("interleaved sequences", () => {

      it("handles alternating consume-refund pattern without drift", () => {
        Arrange: const budget = new IterationBudget(3)
        Act:     10 cycles of [consume, refund]
        Assert:
          Each consume returns true
          expect(budget.used).toBe(0)
          expect(budget.remaining).toBe(3)
      });

      it("tracks state correctly through complex sequence", () => {
        Arrange: const budget = new IterationBudget(5)
        Act + Assert at each step:
          consume x3 -> expect(budget.used).toBe(3)
          refund  x1 -> expect(budget.used).toBe(2)
          consume x2 -> expect(budget.used).toBe(4)
          refund  x4 -> expect(budget.used).toBe(0)
          consume x5 -> expect(budget.used).toBe(5)
          consume x1 -> expect(result).toBe(false)
          refund  x1 -> expect(budget.used).toBe(4)
          consume x1 -> expect(result).toBe(true)
                      -> expect(budget.used).toBe(5)
      });
    });
  });
