  describe("IterationBudget", () => {

    --- 1.1 Constructor & Defaults ---

    describe("constructor", () => {

      it("creates a budget with specified maxTotal and zero used", () => {
        Verify: new IterationBudget(90) has:
          - maxTotal === 90
          - used === 0
          - remaining === 90
        Assertion style: Direct property checks with expect().toBe()
      });

      it("accepts maxTotal of 1 (minimum useful budget)", () => {
        Verify: new IterationBudget(1) has:
          - maxTotal === 1
          - remaining === 1
          - consume() returns true
          - consume() returns false (budget exhausted after 1)
      });

      it("handles maxTotal of 0 (immediately exhausted)", () => {
        Verify: new IterationBudget(0) has:
          - maxTotal === 0
          - remaining === 0
          - consume() returns false immediately
        Edge case: A zero-budget should be immediately exhausted.
        This prevents runaway loops if config accidentally provides 0.
      });
    });

    --- 1.2 consume() Method ---

    describe("consume", () => {

      it("returns true and increments used when under budget", () => {
        Setup: budget = new IterationBudget(5)
        Action: Call consume()
        Verify:
          - Returns true
          - budget.used === 1
          - budget.remaining === 4
        Hermes analog: test_iteration_budget_consume_returns_false_when_exhausted
      });

      it("returns true for each consume up to maxTotal", () => {
        Setup: budget = new IterationBudget(3)
        Action: Call consume() 3 times
        Verify:
          - All 3 return true
          - budget.used === 3
          - budget.remaining === 0
      });

      it("returns false when budget is exhausted", () => {
        Setup: budget = new IterationBudget(3)
        Action: Consume 3 times (exhaust), then consume once more
        Verify:
          - 4th consume returns false
          - budget.used === 3 (unchanged)
          - budget.remaining === 0 (unchanged)
        Hermes analog: test_iteration_budget_consume_returns_false_when_exhausted
      });

      it("returns false on repeated calls after exhaustion", () => {
        Setup: budget = new IterationBudget(2)
        Action: Consume 2 (exhaust), then consume 3 more times
        Verify: All 3 extra calls return false; used stays at 2
        Edge case: Verify consume is idempotent when exhausted.
      });
    });

    --- 1.3 refund() Method ---

    describe("refund", () => {

      it("decrements used count by 1", () => {
        Setup: budget = new IterationBudget(5), consume twice
        Action: refund()
        Verify:
          - budget.used === 1
          - budget.remaining === 4
        Hermes analog: test_iteration_budget_refund_restores_consume
      });

      it("restores consume ability after exhaustion", () => {
        Setup: budget = new IterationBudget(2), consume 2 (exhausted)
        Action: refund()
        Verify:
          - budget.remaining === 1
          - consume() returns true
        Hermes analog: test_iteration_budget_refund_restores_consume
      });

      it("floors used at 0 (refund on fresh budget is no-op)", () => {
        Setup: budget = new IterationBudget(5)
        Action: refund() (without any prior consume)
        Verify:
          - budget.used === 0 (not -1)
          - budget.remaining === 5 (unchanged)
        Edge case: Prevents underflow. Hermes clamps to floor 0.
      });

      it("floors used at 0 after multiple refunds beyond usage", () => {
        Setup: budget = new IterationBudget(3), consume once
        Action: refund() 5 times
        Verify:
          - budget.used === 0 (never goes negative)
          - budget.remaining === 3
      });
    });

    --- 1.4 used / remaining Properties ---

    describe("used and remaining", () => {

      it("used reflects consume and refund accurately", () => {
        Setup: budget = new IterationBudget(5)
        Action sequence:
          - Initial: used=0
          - consume(): used=1
          - consume(): used=2
          - refund(): used=1
          - consume(): used=2
          - refund(): used=1
          - refund(): used=0
        Verify: Each step matches expected used count
        Hermes analog: test_iteration_budget_used_reflects_consume_and_refund
      });

      it("remaining equals maxTotal minus used", () => {
        Setup: budget = new IterationBudget(5)
        Action sequence:
          - consume 3: remaining=2
          - refund 1: remaining=3
          - consume 3: remaining=0
        Verify: Each step matches expected remaining count
        Hermes analog: test_iteration_budget_remaining
      });

      it("remaining never goes below 0", () => {
        Setup: budget = new IterationBudget(2)
        Action: consume 2, then try consume 5 more
        Verify: remaining stays at 0 after exhaustion
      });
    });

    --- 1.5 Interleaved Consume/Refund Sequences ---

    describe("interleaved sequences", () => {

      it("handles alternating consume-refund pattern without drift", () => {
        Setup: budget = new IterationBudget(3)
        Action: 10 cycles of [consume, refund]
        Verify:
          - Each consume returns true
          - After all cycles: used=0, remaining=3
        Validates no drift from alternating operations.
      });

      it("tracks state correctly through complex consume/refund sequence", () => {
        Setup: budget = new IterationBudget(5)
        Action:
          - consume x3 -> used=3
          - refund x1 -> used=2
          - consume x2 -> used=4
          - refund x4 -> used=0
          - consume x5 -> used=5 (exhausted)
          - consume x1 -> false, used=5
          - refund x1 -> used=4
          - consume x1 -> true, used=5
        Verify: Each intermediate state
        Purpose: Exercises the full API surface in a realistic scenario
        matching the run.ts loop (consume per iteration, refund on
        compaction restart, re-consume on retry).
      });
    });

    --- 1.6 Snapshot Method (if implemented per plan) ---

    describe("snapshot", () => {

      it("returns frozen state object with used, remaining, maxTotal", () => {
        Setup: budget = new IterationBudget(10), consume 3
        Verify: budget.snapshot() returns { used: 3, remaining: 7, maxTotal: 10 }

These test cases are mapped to specific issues and edge cases identified

--- R1: Runaway Loop Prevention (Primary Feature Justification) ---

  File: run.iteration-budget.test.ts
  Test: "budget prevents unbounded loop iterations"
  Scenario:
    - Config maxIterations=5
    - Mock: LLM returns tool calls indefinitely (never text)
    - Verify: Loop exits after exactly 5 iterations
    - Verify: Summary or error produced, not infinite hang
  Why: This is THE core value proposition. Multiple OpenClaw issues document
       runaway loops and OOM kills. Budget provides a hard stop.

--- R2: Volatile MessageId Loop Detection Bypass ---

  File: run.iteration-budget.test.ts
  Test: "budget stops loop even when tool-loop-detection is defeated by volatile IDs"
  Scenario:
    - Config: iterationBudget enabled, tool-loop-detection disabled (default)
    - Mock: LLM calls same tool repeatedly with volatile messageIds in results
    - Verify: Budget exhaustion fires at maxIterations regardless of loop detection state
  Why: Tool-loop-detection uses content hashing that fails with volatile IDs (#89090).
       Budget provides an independent counting guard.

--- R3: Budget Resets on Failover ---

  File: run.iteration-budget.test.ts
  Test: "fresh budget on failover model retry"
  Scenario:
    - Config maxIterations=5
    - Mock: 3 iterations, then FailoverError → retry with new model
    - Verify: New model gets a fresh budget (5 iterations), not residual (2)
  Why: FIX #5. Failover creates a new runEmbeddedAgentInternal call with
       fresh IterationBudget.

--- R4: Budget Coexists with MAX_RUN_LOOP_ITERATIONS ---

  File: run.iteration-budget.test.ts
  Test: already covered in 4.9 "budget and MAX_RUN_LOOP_ITERATIONS are independent"
  Why: Budget (productive iterations) and retry limit (error retries) are
       orthogonal guards. Neither should shadow the other.

--- R5: Stale Retry Instruction Cleanup ---

  File: run.iteration-budget.test.ts
  Test: already covered in 4.4 "clears stale retry instructions"
  Why: FIX #4. Without cleanup, summary prompt contains conflicting instructions.

--- R6: Budget with Tool Loop Detection (Both Enabled) ---

  File: run.iteration-budget.test.ts
  Test: "budget and tool-loop-detection coexist without interference"
  Scenario:
    - Config: Both iterationBudget.enabled=true AND tools.loopDetection.enabled=true
    - Mock: LLM makes repeated identical tool calls
    - Verify: Tool-loop-detection fires its warning/block first (lower threshold)
    - Verify: Budget is still consumed correctly even when tool-loop blocks tools
  Why: Both guards operate independently. Budget counts loop iterations;
       tool-loop-detection analyzes call patterns.

   IterationBudget does NOT need a Lock/Mutex. However, async re-entrancy
   should still be considered (e.g., concurrent promise resolution).

   (e.g., import "./iteration-budget.js"). All test imports must follow
   this convention.

4. MOCK MODULE RESOLUTION: Tests using vi.doMock() must be careful about
   module resolution order. Follow the pattern in
   run.overflow-compaction.harness.ts for correct vi.doMock() placement.

5. TEST ISOLATION: Each test should be independent. Use beforeEach() to
   reset mocks and create fresh instances. Never share mutable state
   between tests.

6. FIXTURE PATTERN: Follow OpenClaw's makeAttemptResult() pattern for
   constructing mock attempt results. Use spread operators for overrides:
   makeAttemptResult({ ...defaults, promptError: null, toolMetas: [...] })
