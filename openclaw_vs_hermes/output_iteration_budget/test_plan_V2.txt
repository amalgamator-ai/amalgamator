Purpose:      Unit tests for the IterationBudget class
Source file:  src/agents/iteration-budget.ts (new file, Step 1)
Hermes analog: tests/run_agent/test_iteration_budget_race.py

--- Why no thread-safety tests ---

  Hermes's test_iteration_budget_used_is_thread_safe uses 10 threads x 200
  ops with ThreadPoolExecutor to verify the threading.Lock protects
  concurrent access. Node.js is single-threaded (no data races possible),
  so the TypeScript IterationBudget does NOT need a Lock/Mutex and the
  thread-safety test is correctly EXCLUDED.

--- Imports ---

  import { describe, it, expect } from "vitest";
  import { IterationBudget } from "./iteration-budget.js";

--- Test Structure ---

  describe("IterationBudget", () => {

    // === 1.1 Constructor & Defaults ===

    describe("constructor", () => {

      it("creates a budget with specified maxTotal and zero used", () => {
        Arrange: const budget = new IterationBudget(90)
        Assert:
          expect(budget.maxTotal).toBe(90)
          expect(budget.used).toBe(0)
          expect(budget.remaining).toBe(90)
      });

      it("accepts maxTotal of 1 (minimum useful budget)", () => {
        Arrange: const budget = new IterationBudget(1)
        Assert:
          expect(budget.maxTotal).toBe(1)
          expect(budget.remaining).toBe(1)
          expect(budget.consume()).toBe(true)
          expect(budget.consume()).toBe(false)
      });

      it("handles maxTotal of 0 (immediately exhausted)", () => {
        Arrange: const budget = new IterationBudget(0)
        Assert:
          expect(budget.maxTotal).toBe(0)
          expect(budget.remaining).toBe(0)
          expect(budget.consume()).toBe(false)
      });
    });

    // === 1.2 consume() ===

    describe("consume", () => {

      it("returns true and increments used when under budget", () => {
        Arrange: const budget = new IterationBudget(5)
        Act:     const result = budget.consume()
        Assert:
          expect(result).toBe(true)
          expect(budget.used).toBe(1)
          expect(budget.remaining).toBe(4)
      });

      it("returns true for each consume up to maxTotal", () => {
        Arrange: const budget = new IterationBudget(3)
        Act:     const results = [budget.consume(), budget.consume(), budget.consume()]
        Assert:
          expect(results).toEqual([true, true, true])
          expect(budget.used).toBe(3)
          expect(budget.remaining).toBe(0)
      });

      it("returns false when budget is exhausted", () => {
        Arrange: const budget = new IterationBudget(3)
        Act:     consume 3 times, then const fourth = budget.consume()
        Assert:
          expect(fourth).toBe(false)
          expect(budget.used).toBe(3)   // unchanged
          expect(budget.remaining).toBe(0)
      });

      it("returns false on repeated calls after exhaustion", () => {
        Arrange: const budget = new IterationBudget(2)
        Act:     consume 2 (exhaust), then consume 3 more
        Assert:
          All 3 extra calls return false
          expect(budget.used).toBe(2)   // stays at 2
      });
    });

    // === 1.3 refund() ===

    describe("refund", () => {

      it("decrements used count by 1", () => {
        Arrange: const budget = new IterationBudget(5)
        Act:     budget.consume(); budget.consume(); budget.refund()
        Assert:
          expect(budget.used).toBe(1)
          expect(budget.remaining).toBe(4)
      });

      it("restores consume ability after exhaustion", () => {
        Arrange: const budget = new IterationBudget(2)
        Act:     consume 2 (exhausted), then refund()
        Assert:
          expect(budget.remaining).toBe(1)
          expect(budget.consume()).toBe(true)
      });

      it("floors used at 0 (refund on fresh budget is no-op)", () => {
        Arrange: const budget = new IterationBudget(5)
        Act:     budget.refund()  // refund without prior consume
        Assert:
          expect(budget.used).toBe(0)        // not -1
          expect(budget.remaining).toBe(5)   // unchanged
      });

      it("floors used at 0 after multiple refunds beyond usage", () => {
        Arrange: const budget = new IterationBudget(3)
        Act:     budget.consume(); for (let i = 0; i < 5; i++) budget.refund()
        Assert:
          expect(budget.used).toBe(0)        // never goes negative
          expect(budget.remaining).toBe(3)
      });
    });

    // === 1.4 used / remaining Properties ===

    describe("used and remaining", () => {

      it("used reflects consume and refund accurately", () => {
        Mirrors Hermes test_iteration_budget_used_reflects_consume_and_refund.

        Arrange: const budget = new IterationBudget(10)
        Act + Assert:
          expect(budget.used).toBe(0)
          budget.consume(); expect(budget.used).toBe(1)
          budget.consume(); expect(budget.used).toBe(2)
          budget.refund();  expect(budget.used).toBe(1)
          budget.refund();  expect(budget.used).toBe(0)
      });

      it("remaining equals maxTotal minus used", () => {
        Mirrors Hermes test_iteration_budget_remaining.

        Arrange: const budget = new IterationBudget(5)
        Act + Assert:
          expect(budget.remaining).toBe(5)
          budget.consume(); expect(budget.remaining).toBe(4)
          budget.consume(); budget.consume(); expect(budget.remaining).toBe(2)
          budget.refund(); expect(budget.remaining).toBe(3)
      });

      it("remaining never goes below 0", () => {
        Arrange: const budget = new IterationBudget(2)
        Act:     consume 2, then try consume 5 more
        Assert:  expect(budget.remaining).toBe(0) after each extra consume
      });
    });

    // === 1.5 Interleaved Consume/Refund Sequences ===

    describe("interleaved sequences", () => {

      it("handles alternating consume-refund pattern without drift", () => {
        Arrange: const budget = new IterationBudget(3)
        Act:     10 cycles of [consume, refund]
        Assert:
          Each consume returns true (budget is never exhausted)
          expect(budget.used).toBe(0)
          expect(budget.remaining).toBe(3)
      });

      it("tracks state correctly through complex sequence", () => {
        Arrange: const budget = new IterationBudget(5)
        Act + Assert at each step:
          consume x3 -> expect(budget.used).toBe(3)
          refund  x1 -> expect(budget.used).toBe(2)
          consume x2 -> expect(budget.used).toBe(4)
          refund  x4 -> expect(budget.used).toBe(0)     // floors at 0
          consume x5 -> expect(budget.used).toBe(5)     // exhausted
          consume x1 -> expect(result).toBe(false)       // rejected
          refund  x1 -> expect(budget.used).toBe(4)
          consume x1 -> expect(result).toBe(true)        // restored
                      -> expect(budget.used).toBe(5)
      });
    });

    NOTE: NO snapshot() method tests -- IterationBudget has only:
    constructor, consume, refund, used (property), remaining (property).
    Step 13 reads used/remaining directly for result metadata.
  });

--- Why EXTEND, not NEW ---

  OpenClaw convention: one test file per schema file. All 30+ zod-schema
  test files match their source file name. The pattern is:
    zod-schema.agent-defaults.ts -> zod-schema.agent-defaults.test.ts
  Creating a separate zod-schema.iteration-budget.test.ts would violate
  this convention.

--- No New Imports Needed ---

  The existing file already imports AgentDefaultsSchema and uses:
    expectSchemaSuccess(result: SafeParseReturnType<...>)
    expectSchemaFailurePath(result: SafeParseReturnType<...>, path: string)

  These helpers are defined INSIDE the test file (not imported):
    - expectSchemaSuccess: asserts result.success === true
    - expectSchemaFailurePath: asserts !result.success and checks that
      at least one issue path matches the expected prefix

--- Why NO default value tests here ---

  AgentDefaultsSchema uses .optional() on all fields, NOT .default().
  Calling .parse({iterationBudget: {enabled: true}}) returns the input
  as-is WITHOUT filling in maxIterations=90. Default merging happens in
  resolveAgentConfig() (tested in FILE 3). This was V1 FIX #3/#6.

  // Budget config factory -- small budgets for fast tests
  function makeBudgetConfig(overrides?: Partial<IterationBudgetConfig>) {
    return {
      iterationBudget: {
        enabled: true,
        maxIterations: 10,
        subagentMaxIterations: 5,
        forceSummaryOnExhaustion: true,
        ...overrides,
      },
    };
  }

  // Attempt that produces tool calls but no text payload
  // payloadCount = 0 because assistantTexts is empty
  function makeToolCallAttempt(toolCount = 1) {
    return makeAttemptResult({
      promptError: null,
      assistantTexts: [],     // no text payload -> payloadCount = 0
      toolMetas: Array.from({ length: toolCount }, (_, i) => ({
        toolName: `tool-${i}`,
      })),
    });
  }

  // Attempt that produces text payload (successful completion)
  // payloadCount >= 1 because assistantTexts has content
  function makePayloadAttempt(text = "done") {
    return makeAttemptResult({
      promptError: null,
      assistantTexts: [text],
    });
  }

  // Attempt that produces no text and no tools (empty response)
  function makeEmptyAttempt() {
    return makeAttemptResult({
      promptError: null,
      assistantTexts: [],
      toolMetas: [],
    });
  }

================================================================================
CROSS-CUTTING: REGRESSION SCENARIOS
================================================================================

These regression scenarios map to known issues and are covered by the
tests above. This section documents the mapping.

--- R1: Runaway Loop Prevention ---
  Issue: Agent enters infinite tool-calling loop, OOM kills, unbounded
         token spend.
  Test:  FILE 4, section 4.3 "triggers budget exhaustion when budget
         runs out with no payload"
  Guard: Budget provides a hard iteration count limit regardless of
         tool-loop-detection heuristics.

--- R2: Volatile MessageId Loop Detection Bypass ---
  Issue: Messaging tool results contain volatile IDs (messageId,
         timestamps) that defeat content-hash dedup in tool-loop-detection.
  Test:  FILE 4, section 4.10 "budget stops loop even when
         tool-loop-detection is defeated"
  Guard: Budget counting is independent of content hashing -- it counts
         iterations, not content signatures.

--- R3: Budget Resets on Failover ---
  Issue: If a FailoverError triggers mid-run, budget state may be
         inconsistent.
  Behavior: Budget RESETS on failover. FailoverError propagates up from
            the while(true) loop. The outer wrapper catches it and retries
            with a different model, creating a NEW IterationBudget.
  Test:  Difficult to test via the harness because failover happens in
         the outer runEmbeddedAgent wrapper. Would need to simulate a
         full failover cycle. Consider deferring to E2E testing.
  NOTE:  A fresh budget for the failover model is acceptable because
         pre-failover iterations were wasted (error scenarios).

--- R4: Budget Coexists with MAX_RUN_LOOP_ITERATIONS ---
  Issue: Budget must not interfere with the existing retry-storm guard.
  Test:  FILE 4, section 4.9 "budget and MAX_RUN_LOOP_ITERATIONS are
         independent guards"
  Guard: Budget is a configurable turn limit; MAX_RUN_LOOP_ITERATIONS is
         a defensive retry cap. They operate independently.

--- R5: Stale Retry Instruction Cleanup ---
  Issue: Budget exhaustion summary attempt has conflicting prompt
         instructions from previous retry iterations.
  Test:  FILE 4, section 4.4 "clears stale retry instructions before
         summary attempt"
  Guard: Budget exhaustion handler clears reasoningOnlyRetryInstruction,
         emptyResponseRetryInstruction, compactionContinuationRetryInstruction
         before continuing with summary instruction.

--- R6: Budget with Tool Loop Detection (Both Enabled) ---
  Issue: Budget counting and tool-loop-detection operating simultaneously
         might interfere.
  Test:  With Step 14 changing the default to enabled=true, BOTH guards
         are on by default. This is the default case for most tests in
         FILE 4 (V1 FIX #19).
  Guard: Budget counting and loop detection are independent subsystems.
         Budget counts iterations; loop detection checks content patterns.

--- R7: No Intermediate Pressure Warnings ---
  Design: Hermes explicitly chose NOT to emit "you're running low on
          budget" warnings before exhaustion (agent_init.py lines 530-535):
          "No intermediate pressure warnings -- they caused models to
          'give up' prematurely on complex tasks (#7915)."
  Test:   Not explicitly tested. The IterationBudget class has no
          warning mechanism. Verified by absence of warning code in
          implementation.
