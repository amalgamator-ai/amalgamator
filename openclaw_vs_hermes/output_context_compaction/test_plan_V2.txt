================================================================================

  FRAMEWORK:     Vitest (matches OpenClaw's existing test infrastructure)
  RUNNER CONFIG: Existing vitest shard configs; new tests slot into shards
                 per Section 15.
  CO-LOCATION:   Test files placed adjacent to source files (OpenClaw
                 convention)
  NAMING:        {module}.test.ts for unit/integration
  MOCKING:       vi.hoisted() for module-level mocks, vi.fn() for function
                 stubs, vi.mock() with vi.importActual() for partial module
                 overrides, testing export pattern for dependency injection.

  COVERAGE GOALS:
    - Every new exported function has at least one happy-path and one error test
    - Every Hermes regression test that has a corresponding concept in the port
      has a mapped test case in OpenClaw
    - Every corrected property path (FIX-1 through FIX-5 in port plan Review V4)
      has an explicit assertion verifying the correct path
    - Configuration defaults and boundary values are tested
    - Abort/timeout propagation is tested at every async boundary
    - Zod schema .strict() compliance verified for new config fields

  DESIGN PRINCIPLES:
    DP-1: Test behavior, not implementation. Mock at module boundaries
          (LLM calls, contextEngine.compact, session I/O), not internal
          helpers.
    DP-2: Use OpenClaw's existing fixture factories (makeAgentAssistantMessage,
          makeAgentUserMessage, castAgentMessage from
          src/agents/test-helpers/agent-message-fixtures.ts) wherever possible.
          Only add new helpers for genuinely new patterns (ToolResultMessage
          construction, conversation-with-tool-calls builders).
    DP-3: Token estimation conventions:
          - Suite C (agent-core package): Uses REAL estimateTokens function
            (deterministic char/4 heuristic). Tests construct messages with
            specific text sizes (e.g., "x".repeat(4*N) = N tokens) to
            control the token budget and force a known initial cut point.
          - All other suites: Mock token counts where needed, or use the
            same char/4 formula for consistency.
    DP-4: Each test suite is independently runnable. No cross-file state.
    DP-5: No external service dependencies. All LLM calls are mocked.
    DP-6: Flat describe blocks at top level (matching OpenClaw convention).
          Nested describes allowed for logical sub-grouping within a file.
    DP-7: Reuse existing harness infrastructure. New integration tests extend
          run.overflow-compaction.harness.ts rather than building a new harness.
    DP-8: Pure functions (no side effects) need no mocks — test with
          real inputs and verify outputs directly.

    import type { ToolResultMessage, TextContent } from
      "openclaw/plugin-sdk/llm";

    /**
     * Creates a ToolResultMessage with a single TextContent block.
     * Uses correct ToolResultMessage field names: toolCallId, toolName.
     * (NOT ToolCall field names: id, name)
     */
    export function makeToolResultMessage(
      toolCallId: string,
      toolName: string,
      content: string,
      overrides?: Partial<ToolResultMessage>,
    ): ToolResultMessage {
      return {
        role: "toolResult",
        toolCallId,
        toolName,
        content: [{ type: "text", text: content } as TextContent],
        isError: false,
        timestamp: Date.now(),
        ...overrides,
      };
    }

  3.2 ASSISTANT WITH TOOL CALLS BUILDER (new — combines existing factories)

    import { makeAgentAssistantMessage } from "./agent-message-fixtures.js";
    import type { AssistantMessage, ToolCall, TextContent }
      from "openclaw/plugin-sdk/llm";

    /**
     * Creates an AssistantMessage with ToolCall content blocks.
     * Uses correct ToolCall field names: id, name, arguments.
     * (NOT ToolResultMessage field names: toolCallId, toolName)
     * Delegates to makeAgentAssistantMessage for the base structure.
     */
    export function makeAssistantWithToolCalls(
      toolCalls: Array<{
        id: string;
        name: string;
        arguments: Record<string, unknown>;
      }>,
      text?: string,
      timestamp?: number,
    ): AssistantMessage {
      const content: (TextContent | ToolCall)[] = [];
      if (text) {
        content.push({ type: "text", text });
      }
      for (const tc of toolCalls) {
        content.push({
          type: "toolCall",
          id: tc.id,
          name: tc.name,
          arguments: tc.arguments,
        });
      }
      return makeAgentAssistantMessage({
        content,
        timestamp: timestamp ?? Date.now(),
        stopReason: "tool_use",
      });
    }

  3.3 CONVERSATION BUILDERS (new — no existing equivalent)

    import type { AgentMessage } from "openclaw/plugin-sdk/agent-core";

    /**
     * Creates an alternating user/assistant conversation.
     * Each message has `textSize` characters (default 100).
     * Timestamps are sequential starting from 1000.
     */
    export function makeAlternatingConversation(
      turns: number,
      textSize = 100,
    ): AgentMessage[] {
      const messages: AgentMessage[] = [];
      for (let i = 0; i < turns; i++) {
        const ts = 1000 + i * 100;
        const text = String.fromCharCode(65 + (i % 26)).repeat(textSize);
        if (i % 2 === 0) {
          messages.push(makeAgentUserMessage({
            content: [{ type: "text", text }],
            timestamp: ts,
          }));
        } else {
          messages.push(makeAgentAssistantMessage({
            content: [{ type: "text", text }],
            timestamp: ts,
          }));
        }
      }
      return messages;
    }

    /**
     * Creates a realistic conversation with tool use patterns.
     * Each assistant turn has `toolCallsPerTurn` ToolCall blocks.
     * Each tool call is followed by a ToolResultMessage.
     */
    export function makeConversationWithToolCalls(
      userTurns: number,
      toolCallsPerTurn: number,
      argSize = 100,
    ): AgentMessage[] {
      const messages: AgentMessage[] = [];
      let ts = 1000;
      for (let i = 0; i < userTurns; i++) {
        // User message
        messages.push(makeAgentUserMessage({
          content: [{ type: "text", text: `User turn ${i}` }],
          timestamp: ts++,
        }));
        // Assistant with tool calls
        const calls = Array.from({ length: toolCallsPerTurn }, (_, j) => ({
          id: `tc-${i}-${j}`,
          name: `tool_${j}`,
          arguments: { input: "x".repeat(argSize) },
        }));
        messages.push(makeAssistantWithToolCalls(calls, undefined, ts++));
        // Tool results
        for (const call of calls) {
          messages.push(makeToolResultMessage(
            call.id, call.name,
            `Result for ${call.name}: ${"y".repeat(argSize)}`,
            { timestamp: ts++ },
          ));
        }
      }
      return messages;
    }

  3.4 SESSION TREE ENTRY BUILDERS (new — for agent-core package tests)

    import type { AgentMessage } from "openclaw/plugin-sdk/agent-core";
    import type { SessionTreeEntry } from
      "packages/agent-core/src/harness/types.js";

    /**
     * Wraps an AgentMessage in a MessageEntry SessionTreeEntry.
     * Matches the exact pattern from compaction-trailing-toolresult.test.ts
     * and compaction-image-tokens.test.ts:
     *   - id: `entry-${index}`
     *   - parentId: null for first entry, `entry-${index-1}` for rest
     *   - timestamp: ISO string from message.timestamp
     *   - message: the AgentMessage itself
     */
    export function messageEntry(
      message: AgentMessage,
      index: number,
    ): SessionTreeEntry {
      return {
        type: "message",
        id: `entry-${index}`,
        parentId: index === 0 ? null : `entry-${index - 1}`,
        timestamp: new Date(message.timestamp).toISOString(),
        message,
      };
    }

    /**
     * Convenience: maps an array of AgentMessages through messageEntry
     * with sequential indices.
     */
    export function buildSessionTree(
      messages: AgentMessage[],
    ): SessionTreeEntry[] {
      return messages.map((msg, i) => messageEntry(msg, i));
    }
