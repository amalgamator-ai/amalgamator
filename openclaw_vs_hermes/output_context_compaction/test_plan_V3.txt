1. TESTING STRATEGY OVERVIEW
================================================================================

  PRINCIPLES:

  P1: Follow OpenClaw conventions exactly — Vitest, co-located *.test.ts,
      describe/it blocks, vi.fn()/vi.mock()/vi.doMock() for mocking.

  P2: Pure functions get pure tests — no mocks needed for
      compressionMadeProgress or shouldContinueMultiPass.

  P3: Injected dependencies get direct mocks — multiPassCompact receives
      a contextEngine (or compact function) as a parameter. Tests create
      mock objects with vi.fn() and inject them directly. No module
      mocking needed.

  P4: Module-level mocking for integration tests only — Suite F and
      Suite I use vi.doMock() via the existing harness extension pattern
      to intercept module imports.

  P5: Existing factories FIRST — reuse makeAgentUserMessage,
      makeAgentAssistantMessage, makeCompactionSuccess from existing
      test helpers. Only create new factories for new types.

  P6: Test isolation — each test is independent. beforeEach resets all
      mocks. No shared mutable state across tests.

  P7: Shard alignment — test files placed in directories matching
      the existing vitest shard configuration (verified against
      vitest.agents-paths.mjs and vitest.*.config.ts files).

  COVERAGE TARGETS:

  - Every new public function has unit tests
  - Every configuration field has defaults + validation tests
  - Every error path has a test
  - Every Hermes regression concern has a mapped test
  - Every port plan FIX has a verifying test

  TOTAL PLANNED TESTS: 110 across 9 suites

4. TEST SUITE A — PROGRESS DETECTION & ANTI-THRASHING
================================================================================

  FILE: src/agents/agent-hooks/compaction-progress.test.ts
  SOURCE: src/agents/agent-hooks/compaction-progress.ts (NEW)
  PORTED FROM: Hermes agent/turn_context.py _compression_made_progress
  SHARD: agents-support

  FRAMEWORK SETUP:

    import { describe, expect, it } from "vitest";
    import {
      compressionMadeProgress,
      shouldContinueMultiPass,
    } from "./compaction-progress.js";

  NOTES:
    - compressionMadeProgress: PURE FUNCTION, 4 plain parameters:
      (origLen: number, newLen: number, origTokens: number, newTokens: number)
      Returns boolean. No I/O, no side effects, no mocks needed.
    - shouldContinueMultiPass: Takes a structured params object with
      passNumber, maxPasses, tokensAfter, targetTokens,
      consecutiveNoProgress, and optional abortSignal.
      Returns { continue: boolean; reason?: string }.

  -- describe("compressionMadeProgress") -----------------------------------------

    4.1 it("returns true when message count is reduced")
        CALL: compressionMadeProgress(10, 5, 1000, 1000)
        EXPECT: true
        WHY: Row/message reduction alone counts as progress.
             Ported from Hermes test_rows_reduced_counts_as_progress.

    4.2 it("returns true when tokens are reduced without message change")
        CALL: compressionMadeProgress(220, 220, 288_028, 183_180)
        EXPECT: true
        WHY: Token-only progress is valid (issue #39548). Summarization
             without row removal should not be flagged as "stuck."
             Ported from Hermes test_tokens_reduced_without_row_change.

    4.3 it("returns true when both messages and tokens are reduced")
        CALL: compressionMadeProgress(220, 180, 288_028, 150_000)
        EXPECT: true
        WHY: Common case — summarize + drop messages.

    4.4 it("returns false when neither messages nor tokens changed")
        CALL: compressionMadeProgress(10, 10, 1000, 1000)
        EXPECT: false
        WHY: Genuine stuck condition. No progress on any dimension.

    4.5 it("returns false when both messages and tokens increased")
        CALL: compressionMadeProgress(10, 12, 1000, 1200)
        EXPECT: false
        WHY: Expansion is clearly not progress.

    4.6 it("returns true when messages grew but tokens dropped significantly")
        CALL: compressionMadeProgress(10, 11, 1000, 600)
        EXPECT: true
        WHY: Token reduction > 5% counts even if message count grew.
             Ported from Hermes test_rows_grew_but_tokens_dropped.

    4.7 it("returns true when tokens grew but messages dropped significantly")
        CALL: compressionMadeProgress(10, 5, 1000, 1100)
        EXPECT: true
        WHY: Message reduction alone counts.
             Ported from Hermes test_tokens_grew_but_rows_dropped.

    4.8 it("returns false when token reduction is below 5% threshold")
        CALL A: compressionMadeProgress(10, 10, 1000, 970)
        EXPECT A: false (3% drop — below 5% threshold)
        CALL B: compressionMadeProgress(10, 10, 1000, 940)
        EXPECT B: true (6% drop — above 5% threshold)
        WHY: Noise filter. Sub-5% token wobble is not meaningful progress.
             Ported from Hermes test_sub_5pct_token_drop_is_not_progress.

    4.9 it("returns false when original tokens are zero (degenerate)")
        CALL: compressionMadeProgress(10, 10, 0, 0)
        EXPECT: false
        WHY: Division guard. 0 original tokens means no baseline to measure.
             Ported from Hermes test_zero_orig_tokens_is_not_progress.

  -- describe("compressionMadeProgress edge cases") --------------------------------

    4.10 it("handles NaN token values gracefully")
         CALL: compressionMadeProgress(10, 10, NaN, 50000)
         EXPECT: false
         WHY: NaN propagation guard (TypeScript-specific concern).

    4.11 it("handles negative token values")
         CALL: compressionMadeProgress(10, 10, -100, 50000)
         EXPECT: false
         WHY: Degenerate estimate guard.

    4.12 it("returns false for exactly 5% token reduction (strict < threshold)")
         WHAT: Hermes uses `new_tokens < orig_tokens * 0.95`. The strict
               less-than means 950 < 950 is FALSE. Exactly 5% reduction
               does NOT count as progress — you need MORE than 5%.
         CALL A: compressionMadeProgress(10, 10, 1000, 950)
         EXPECT A: false (exactly 5% reduction — NOT progress)
         CALL B: compressionMadeProgress(10, 10, 1000, 949)
         EXPECT B: true (5.1% reduction — IS progress)
         WHY: Critical boundary test. V2 initially got this WRONG
              (stated "returns true"). 950 < 950 is unambiguously false
              in both Python and TypeScript. This test pins the exact
              boundary behavior.

  -- describe("shouldContinueMultiPass") -------------------------------------------

    DESIGN: shouldContinueMultiPass does NOT receive progressMade directly.
    It receives a consecutiveNoProgress counter maintained by the caller
    (multiPassCompact). The function's decision tree:

      1. Check abortSignal.aborted → false, "aborted"
      2. Check passNumber >= maxPasses → false, "max_passes_reached"
      3. Check tokensAfter <= targetTokens → false, "target_reached"
      4. Check consecutiveNoProgress >= 2 → false, "anti_thrash"
      5. Otherwise → true (continue)

    This design allows the loop to tolerate ONE marginal pass before
    stopping — an INTENTIONAL enhancement over Hermes (which stops
    immediately on any sub-threshold pass).

    4.13 it("returns false when maxPasses reached")
         INPUT: { passNumber: 3, maxPasses: 3,
                  tokensAfter: 120000, targetTokens: 100000,
                  consecutiveNoProgress: 0 }
         EXPECT: { continue: false, reason: "max_passes_reached" }

    4.14 it("returns false when tokens are at or below target")
         INPUT: { passNumber: 1, maxPasses: 3, tokensAfter: 80000,
                  targetTokens: 100000, consecutiveNoProgress: 0 }
         EXPECT: { continue: false, reason: "target_reached" }

    4.15 it("returns true when counter is below anti-thrash threshold")
         WHAT: A single no-progress pass does NOT stop the loop. The
               consecutiveNoProgress counter must reach 2 for anti-thrash
               exit. With counter=1, the function returns true.
         INPUT: { passNumber: 1, maxPasses: 5,
                  tokensAfter: 120000, targetTokens: 100000,
                  consecutiveNoProgress: 1 }
         EXPECT: { continue: true }
         WHY: This is the key INTENTIONAL enhancement over Hermes.
              Hermes breaks immediately; OpenClaw gives a second chance.

    4.16 it("returns true when tokens above target and counter is zero")
         WHAT: Happy path — keep going.
         INPUT: { passNumber: 1, maxPasses: 3,
                  tokensAfter: 120000, targetTokens: 100000,
                  consecutiveNoProgress: 0 }
         EXPECT: { continue: true }

    4.17 it("returns false on abort signal")
         SETUP: const ac = new AbortController(); ac.abort();
         INPUT: { passNumber: 1, maxPasses: 3,
                  tokensAfter: 120000, targetTokens: 100000,
                  consecutiveNoProgress: 0, abortSignal: ac.signal }
         EXPECT: { continue: false, reason: "aborted" }

    4.18 it("returns false when consecutive no-progress count reaches 2")
         INPUT: { passNumber: 2, maxPasses: 5,
                  tokensAfter: 120000, targetTokens: 100000,
                  consecutiveNoProgress: 2 }
         EXPECT: { continue: false, reason: "anti_thrash" }

    4.19 it("checks abort before other conditions")
         WHAT: Abort signal has highest priority — checked first even if
               other exit conditions are met.
         SETUP: const ac = new AbortController(); ac.abort();
         INPUT: { passNumber: 5, maxPasses: 3,
                  tokensAfter: 50000, targetTokens: 100000,
                  consecutiveNoProgress: 5, abortSignal: ac.signal }
         EXPECT: { continue: false, reason: "aborted" }
         WHY: Ensures priority ordering of exit conditions.
